<script>

const decorations = [
  "✨", "🎊", "⭐", "🎉", "🎈"
];

function sparkle() {

  const s = document.createElement("div");

  s.className = "sparkle";

  s.textContent =
    decorations[
      Math.floor(Math.random() * decorations.length)
    ];

  s.style.left = Math.random() * 100 + "vw";

  s.style.fontSize =
    (15 + Math.random() * 25) + "px";

  s.style.animationDuration =
    (2 + Math.random() * 3) + "s";

  document.body.appendChild(s);

  setTimeout(() => s.remove(), 5000);
}

function party() {

  document.getElementById("partyMessage").textContent =
    "WOOOOHOOOO!! LET THE MADNESS BEGIN!! 🥳";

  for (let i = 0; i < 50; i++) {
    setTimeout(sparkle, i * 35);
  }
}

function gift() {

  document.getElementById("surprise").style.display =
    "block";

  document.querySelector(".gift").textContent = "🎊";

  for (let i = 0; i < 35; i++) {
    setTimeout(sparkle, i * 40);
  }
}
